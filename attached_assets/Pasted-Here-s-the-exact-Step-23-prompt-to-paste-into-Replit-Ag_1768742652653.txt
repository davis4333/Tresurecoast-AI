import { z } from "zod";

export const InsightsQuerySchema = z.object({
  botPublicKey: z.string().uuid("Invalid bot public key"),
  days: z.coerce.number().int().min(1).max(90).optional().default(30),
});

export type InsightsQuery = z.infer<typeof InsightsQuerySchema>;

export const InsightsResponseSchema = z.object({
  ok: z.literal(true),
  botPublicKey: z.string().uuid(),
  rangeDays: z.number(),
  totals: z.object({
    chats: z.number(),
    missingDataEvents: z.number(),
    leadCaptureTriggered: z.number(),
    redirectClicks: z.number(),
  }),
  topMissingFields: z.array(
    z.object({
      field: z.string(),
      count: z.number(),
    })
  ),
  topTopics: z.array(
    z.object({
      topic: z.string(),
      count: z.number(),
    })
  ),
  sampleQuestions: z.array(
    z.object({
      message: z.string(),
      createdAt: z.string(),
      topic: z.string().optional(),
      missingFields: z.array(z.string()).optional(),
    })
  ),
  suggestions: z.array(
    z.object({
      title: z.string(),
      reason: z.string(),
      suggestedFields: z.array(z.string()),
      exampleValues: z.any().optional(),
    })
  ),
});

export type InsightsResponse = z.infer<typeof InsightsResponseSchema>;

import { NextRequest, NextResponse } from "next/server";
import { auth } from "@clerk/nextjs/server";
import { prisma } from "@/lib/db";
import { InsightsQuerySchema } from "@/lib/admin/insightsSchemas";
import { ZodError } from "zod";

export const dynamic = "force-dynamic";

export async function GET(req: NextRequest) {
  try {
    // Fail-closed if Clerk not configured
    if (!process.env.NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY || !process.env.CLERK_SECRET_KEY) {
      return NextResponse.json(
        { ok: false, error: "server_misconfigured" },
        { status: 503 }
      );
    }

    const { userId } = await auth();
    if (!userId) {
      return NextResponse.json(
        { ok: false, error: "unauthorized" },
        { status: 401 }
      );
    }

    const searchParams = req.nextUrl.searchParams;
    const rawQuery = {
      botPublicKey: searchParams.get("botPublicKey"),
      days: searchParams.get("days"),
    };

    const query = InsightsQuerySchema.parse(rawQuery);
    const { botPublicKey, days } = query;

    const bot = await prisma.bot.findUnique({
      where: { publicKey: botPublicKey },
      select: { id: true },
    });

    if (!bot) {
      return NextResponse.json(
        { ok: false, error: "not_found" },
        { status: 404 }
      );
    }

    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);

    const conversations = await prisma.conversation.findMany({
      where: {
        botId: bot.id,
        createdAt: { gte: cutoffDate },
      },
      select: { id: true },
    });

    const conversationIds = conversations.map((c) => c.id);

    const [
      missingDataEvents,
      leadCaptureEvents,
      redirectClickEvents,
      events,
      messages,
    ] = await Promise.all([
      prisma.dataEvent.count({
        where: {
          conversationId: { in: conversationIds },
          type: "MISSING_DATA",
        },
      }),
      prisma.dataEvent.count({
        where: {
          conversationId: { in: conversationIds },
          type: "LEAD_CAPTURE_TRIGGERED",
        },
      }),
      prisma.dataEvent.count({
        where: {
          conversationId: { in: conversationIds },
          type: "REDIRECT_CLICK",
        },
      }),
      prisma.dataEvent.findMany({
        where: {
          conversationId: { in: conversationIds },
          type: { in: ["MISSING_DATA", "TRUTH_RESPONSE", "TOPIC_DETECTED"] },
        },
        select: {
          type: true,
          payload: true,
          createdAt: true,
        },
        orderBy: { createdAt: "desc" },
        take: 500,
      }),
      prisma.message.findMany({
        where: {
          conversationId: { in: conversationIds },
          role: "user",
        },
        select: {
          content: true,
          createdAt: true,
        },
        orderBy: { createdAt: "desc" },
        take: 25,
      }),
    ]);

    const missingFieldsMap = new Map<string, number>();
    const topicsMap = new Map<string, number>();

    for (const ev of events) {
      const payload = ev.payload as any;

      if (ev.type === "MISSING_DATA" && Array.isArray(payload?.missingFields)) {
        for (const field of payload.missingFields) {
          missingFieldsMap.set(field, (missingFieldsMap.get(field) || 0) + 1);
        }
      }

      if (payload?.topic && typeof payload.topic === "string") {
        topicsMap.set(payload.topic, (topicsMap.get(payload.topic) || 0) + 1);
      }
    }

    const topMissingFields = Array.from(missingFieldsMap.entries())
      .map(([field, count]) => ({ field, count }))
      .sort((a, b) => b.count - a.count)
      .slice(0, 10);

    const topTopics = Array.from(topicsMap.entries())
      .map(([topic, count]) => ({ topic, count }))
      .sort((a, b) => b.count - a.count)
      .slice(0, 10);

    const sampleQuestions = messages.map((m) => ({
      message: m.content,
      createdAt: m.createdAt.toISOString(),
    }));

    const suggestions = generateSuggestions(topMissingFields, topTopics);

    return NextResponse.json({
      ok: true,
      botPublicKey,
      rangeDays: days,
      totals: {
        chats: conversations.length,
        missingDataEvents,
        leadCaptureTriggered: leadCaptureEvents,
        redirectClicks: redirectClickEvents,
      },
      topMissingFields,
      topTopics,
      sampleQuestions,
      suggestions,
    });
  } catch (error) {
    if (error instanceof ZodError) {
      return NextResponse.json(
        { ok: false, error: "validation_error", details: error.errors },
        { status: 400 }
      );
    }

    return NextResponse.json(
      { ok: false, error: "internal_error" },
      { status: 500 }
    );
  }
}

function generateSuggestions(
  missingFields: Array<{ field: string; count: number }>,
  topics: Array<{ topic: string; count: number }>
) {
  const suggestions: Array<{
    title: string;
    reason: string;
    suggestedFields: string[];
    exampleValues?: any;
  }> = [];

  const fieldSuggestions: Record<string, { title: string; fields: string[]; example?: any }> = {
    hours: {
      title: "Add Business Hours",
      fields: ["hours"],
      example: { monday: "9AM-6PM", tuesday: "9AM-6PM" },
    },
    pricing: {
      title: "Add Service Pricing",
      fields: ["services"],
      example: [{ name: "Haircut", price: "$45" }],
    },
    services: {
      title: "Add Service Descriptions",
      fields: ["services"],
      example: [{ name: "Haircut", description: "Classic cut", durationMinutes: 45 }],
    },
    location: {
      title: "Add Business Location",
      fields: ["businessAddress"],
      example: "123 Main St, Fort Pierce, FL",
    },
    contact: {
      title: "Add Contact Information",
      fields: ["businessPhone", "businessEmail"],
      example: { phone: "(555) 123-4567", email: "info@business.com" },
    },
    policies: {
      title: "Add Business Policies",
      fields: ["policies"],
      example: { cancellation: "24 hour notice required" },
    },
  };

  for (const { field, count } of missingFields.slice(0, 5)) {
    const key = field.toLowerCase();
    const suggestion = fieldSuggestions[key];
    if (suggestion) {
      suggestions.push({
        title: suggestion.title,
        reason: `${count} question${count > 1 ? "s" : ""} about ${field}`,
        suggestedFields: suggestion.fields,
        exampleValues: suggestion.example,
      });
    }
  }

  if (topics.length > 0 && topics[0].count > 5) {
    const topTopic = topics[0].topic;
    suggestions.push({
      title: `Expand "${topTopic}" Information`,
      reason: `${topics[0].count} questions about ${topTopic}`,
      suggestedFields: ["services", "hours", "links"],
    });
  }

  return suggestions;
}

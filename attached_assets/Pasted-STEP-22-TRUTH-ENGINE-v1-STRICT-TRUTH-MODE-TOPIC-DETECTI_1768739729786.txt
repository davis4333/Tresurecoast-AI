import { z } from "zod";

export const TopicSchema = z.enum([
  "HOURS",
  "SERVICES",
  "PRICING",
  "LOCATION",
  "CONTACT",
  "BOOKING",
  "PAYMENT",
  "POLICIES",
  "GENERAL",
]);

export type Topic = z.infer<typeof TopicSchema>;

export type TopicResult = {
  topic: Topic;
  confidence: number; // 0..1
  matched: string[];
};

function hasAny(text: string, words: string[]) {
  return words.filter((w) => text.includes(w));
}

export function detectTopic(message: string): TopicResult {
  const t = message.toLowerCase();

  const hours = hasAny(t, ["hours", "open", "close", "closing", "opening", "when are you open", "what time", "today", "tomorrow"]);
  if (hours.length) return { topic: "HOURS", confidence: 0.9, matched: hours };

  const services = hasAny(t, ["services", "service", "offer", "do you do", "can you do", "haircut", "fade", "beard", "tattoo", "epoxy", "classes"]);
  if (services.length) return { topic: "SERVICES", confidence: 0.75, matched: services };

  const pricing = hasAny(t, ["price", "pricing", "cost", "how much", "rate", "$", "charge"]);
  if (pricing.length) return { topic: "PRICING", confidence: 0.8, matched: pricing };

  const location = hasAny(t, ["address", "located", "location", "where are you", "directions", "near"]);
  if (location.length) return { topic: "LOCATION", confidence: 0.85, matched: location };

  const contact = hasAny(t, ["phone", "call", "number", "email", "contact", "text"]);
  if (contact.length) return { topic: "CONTACT", confidence: 0.85, matched: contact };

  const booking = hasAny(t, ["book", "booking", "schedule", "appointment", "reserve", "availability"]);
  if (booking.length) return { topic: "BOOKING", confidence: 0.85, matched: booking };

  const payment = hasAny(t, ["pay", "payment", "deposit", "invoice", "checkout"]);
  if (payment.length) return { topic: "PAYMENT", confidence: 0.85, matched: payment };

  const policies = hasAny(t, ["policy", "policies", "cancel", "cancellation", "refund", "late", "reschedule"]);
  if (policies.length) return { topic: "POLICIES", confidence: 0.75, matched: policies };

  return { topic: "GENERAL", confidence: 0.4, matched: [] };
}

import type { Bot, BotLink } from "@prisma/client";
import { detectTopic } from "./topics";
import type { TruthResult } from "./types";

type BotWithLinks = Bot & { links: BotLink[] };

function fmtContact(bot: BotWithLinks) {
  const parts: string[] = [];
  if (bot.businessPhone) parts.push(`Phone: ${bot.businessPhone}`);
  if (bot.businessEmail) parts.push(`Email: ${bot.businessEmail}`);
  return parts.length ? parts.join(" • ") : "";
}

function getLink(bot: BotWithLinks, type: "BOOKING" | "PAYMENT") {
  // Prefer BotLink.type match, fallback to any “booking”/“pay” url-ish link in OTHER.
  const byType = bot.links?.find((l) => l.type === type && l.url);
  if (byType?.url) return byType.url;

  const t = type === "BOOKING" ? "book" : "pay";
  const heuristic = bot.links?.find((l) => (l.label || "").toLowerCase().includes(t) && l.url);
  return heuristic?.url || null;
}

function normalizeJson<T>(value: unknown): T | null {
  if (!value) return null;
  try {
    return value as T;
  } catch {
    return null;
  }
}

function renderHours(hours: unknown): { text: string; used: boolean } {
  // Expect array of { day: string, open?: string, close?: string, closed?: boolean }
  const h = normalizeJson<any[]>(hours);
  if (!h || !Array.isArray(h) || h.length === 0) return { text: "", used: false };

  const lines = h
    .map((d) => {
      const day = d.day ?? d.name ?? "";
      if (!day) return null;
      if (d.closed) return `${day}: Closed`;
      const open = d.open ?? d.start ?? "";
      const close = d.close ?? d.end ?? "";
      if (open && close) return `${day}: ${open}–${close}`;
      return `${day}: (hours not fully set)`;
    })
    .filter(Boolean);

  if (!lines.length) return { text: "", used: false };
  return { text: lines.join("\n"), used: true };
}

function renderServices(services: unknown): { text: string; used: boolean; hasPrices: boolean } {
  // Expect array of { name, description?, priceRange?, durationMinutes? }
  const s = normalizeJson<any[]>(services);
  if (!s || !Array.isArray(s) || s.length === 0) return { text: "", used: false, hasPrices: false };

  let hasPrices = false;

  const lines = s
    .map((x) => {
      const name = x.name;
      if (!name) return null;
      const price = x.priceRange || x.price || null;
      if (price) hasPrices = true;
      const dur = x.durationMinutes ? `${x.durationMinutes} min` : x.duration ? `${x.duration}` : null;

      const meta = [price ? `Price: ${price}` : null, dur ? `Duration: ${dur}` : null].filter(Boolean).join(" • ");
      return meta ? `• ${name} — ${meta}` : `• ${name}`;
    })
    .filter(Boolean);

  if (!lines.length) return { text: "", used: false, hasPrices };
  return { text: lines.join("\n"), used: true, hasPrices };
}

export function runTruthEngine(input: { message: string; bot: BotWithLinks }): TruthResult {
  const { message, bot } = input;
  const topicResult = detectTopic(message);

  const missing: string[] = [];
  const sourcedFrom: string[] = [];

  const contactLine = fmtContact(bot);

  // Booking/Payment: prefer redirect flow if links exist
  if (topicResult.topic === "BOOKING") {
    const url = getLink(bot, "BOOKING");
    if (url) {
      sourcedFrom.push("links[BOOKING]");
      return {
        reply: `You can book here: ${url}\n\nIf you’d like, I can also take your details and have the team confirm availability.`,
        intent: "BOOK_OR_PAY_REDIRECT",
        confidence: 0.9,
        sourcedFrom,
        requiresLeadCapture: true,
        missingFields: [],
        topic: topicResult.topic,
      };
    }
    missing.push("links[BOOKING]");
    return {
      reply: `I don’t have a booking link in my system yet. ${contactLine ? `\n\n${contactLine}` : ""}\n\nIf you want, leave your name + best contact and what you’re trying to book — the team will follow up.`,
      intent: "MISSING_DATA",
      confidence: 0.8,
      sourcedFrom,
      requiresLeadCapture: true,
      missingFields: missing,
      topic: topicResult.topic,
    };
  }

  if (topicResult.topic === "PAYMENT") {
    const url = getLink(bot, "PAYMENT");
    if (url) {
      sourcedFrom.push("links[PAYMENT]");
      return {
        reply: `You can pay here: ${url}\n\nIf you want, tell me what service you’re paying for and I’ll pass it to the team.`,
        intent: "BOOK_OR_PAY_REDIRECT",
        confidence: 0.9,
        sourcedFrom,
        requiresLeadCapture: true,
        missingFields: [],
        topic: topicResult.topic

2) Seed data (idempotent-ish):
pnpm prisma db execute --stdin <<'SQL'
-- Ensure at least 2 orgs exist
INSERT INTO "Organization" ("name")
SELECT 'Org A'
WHERE NOT EXISTS (SELECT 1 FROM "Organization" WHERE "name"='Org A');

INSERT INTO "Organization" ("name")
SELECT 'Org B'
WHERE NOT EXISTS (SELECT 1 FROM "Organization" WHERE "name"='Org B');

-- Grab org ids
WITH oa AS (SELECT "id" FROM "Organization" WHERE "name"='Org A' LIMIT 1),
     ob AS (SELECT "id" FROM "Organization" WHERE "name"='Org B' LIMIT 1)
-- Owner in Org A
INSERT INTO "OrganizationMember" ("organizationId","clerkUserId","role","createdAt","updatedAt")
SELECT oa.id, 'user_a_owner', 'AGENCY_OWNER', NOW(), NOW() FROM oa
ON CONFLICT ("organizationId","clerkUserId") DO UPDATE SET "role"=EXCLUDED."role", "updatedAt"=NOW();

WITH oa AS (SELECT "id" FROM "Organization" WHERE "name"='Org A' LIMIT 1)
-- Client in Org A
INSERT INTO "OrganizationMember" ("organizationId","clerkUserId","role","createdAt","updatedAt")
SELECT oa.id, 'user_a_client', 'CLIENT', NOW(), NOW() FROM oa
ON CONFLICT ("organizationId","clerkUserId") DO UPDATE SET "role"=EXCLUDED."role", "updatedAt"=NOW();

WITH ob AS (SELECT "id" FROM "Organization" WHERE "name"='Org B' LIMIT 1)
-- Admin in Org B
INSERT INTO "OrganizationMember" ("organizationId","clerkUserId","role","createdAt","updatedAt")
SELECT ob.id, 'user_b_admin', 'AGENCY_ADMIN', NOW(), NOW() FROM ob
ON CONFLICT ("organizationId","clerkUserId") DO UPDATE SET "role"=EXCLUDED."role", "updatedAt"=NOW();
SQL

3) Show proof:
pnpm prisma db execute --stdin <<'SQL'
SELECT o.id, o.name, m."clerkUserId", m.role
FROM "OrganizationMember" m
JOIN "Organization" o ON o.id = m."organizationId"
ORDER BY o.id, m."clerkUserId";
SQL

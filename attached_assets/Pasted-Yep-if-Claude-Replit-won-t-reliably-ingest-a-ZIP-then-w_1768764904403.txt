import { z } from "zod";

// Strict hex color validation (#RRGGBB)
const hexColorRegex = /^#[0-9A-Fa-f]{6}$/;

// Domain validation (domain only, no protocol)
const domainRegex =
  /^(?:[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?\.)+[a-z0-9][a-z0-9-]{0,61}[a-z0-9]$/i;

export const BrandingSchema = z.object({
  whiteLabelEnabled: z.boolean(),
  brandCompanyName: z.string().max(80).nullable().or(z.literal("")),
  brandLogoUrl: z.string().url().nullable().or(z.literal("")),
  brandPrimaryColor: z
    .string()
    .regex(hexColorRegex, "Must be a valid hex color (#RRGGBB)")
    .nullable()
    .or(z.literal("")),
  customDomain: z
    .string()
    .regex(domainRegex, "Must be a valid domain (e.g., example.com)")
    .nullable()
    .or(z.literal("")),
  showPoweredBy: z.boolean().optional(),
});

export type BrandingInput = z.infer<typeof BrandingSchema>;

export const BrandingResponseSchema = z.object({
  ok: z.literal(true),
  branding: BrandingSchema,
});

export type BrandingResponse = z.infer<typeof BrandingResponseSchema>;
EOF

import { NextRequest, NextResponse } from "next/server";
import { auth } from "@clerk/nextjs/server";
import { prisma } from "@/lib/db";
import { BrandingSchema } from "@/lib/validators/branding";
import { logger } from "@/lib/logger";
import { AppError, createErrorResponse } from "@/lib/errors";
import { ZodError } from "zod";

export const dynamic = "force-dynamic";

// GET /api/org/branding - Get current org branding (any org member)
export async function GET(_req: NextRequest) {
  const requestId = crypto.randomUUID();

  try {
    const { userId } = await auth();
    if (!userId) throw new AppError("unauthorized", "Authentication required", 401);

    const membership = await prisma.organizationMember.findFirst({
      where: { clerkUserId: userId },
      include: {
        organization: {
          select: {
            id: true,
            whiteLabelEnabled: true,
            brandCompanyName: true,
            brandLogoUrl: true,
            brandPrimaryColor: true,
            customDomain: true,
            showPoweredBy: true,
          },
        },
      },
    });

    if (!membership) throw new AppError("not_found", "Organization not found", 404);

    const branding = {
      whiteLabelEnabled: membership.organization.whiteLabelEnabled,
      brandCompanyName: membership.organization.brandCompanyName,
      brandLogoUrl: membership.organization.brandLogoUrl,
      brandPrimaryColor: membership.organization.brandPrimaryColor,
      customDomain: membership.organization.customDomain,
      showPoweredBy: membership.organization.showPoweredBy,
    };

    logger.info({ requestId, userId, organizationId: membership.organizationId }, "Branding retrieved");

    return NextResponse.json({ ok: true, branding });
  } catch (error) {
    if (error instanceof AppError) return createErrorResponse(error, requestId);
    logger.error({ requestId, error }, "Failed to retrieve branding");
    return createErrorResponse(new AppError("internal_error", "Failed to retrieve branding", 500), requestId);
  }
}
